cmake_minimum_required(VERSION 3.25)
project(hdrbridge_android LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MAP_IMPORTED_CONFIG_RELWITHDEBINFO Release)
get_filename_component(HDRBRIDGE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(ANDROID_ROOT "${HDRBRIDGE_ROOT}/android")
set(DEPS "${ANDROID_ROOT}/vcpkg_installed/arm64-android-hdr")
list(PREPEND CMAKE_PREFIX_PATH "${DEPS}")
list(PREPEND CMAKE_FIND_ROOT_PATH "${DEPS}")
list(APPEND CMAKE_MODULE_PATH "${DEPS}/share/jxr")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${HDRBRIDGE_ROOT}/VERSION")
file(STRINGS "${HDRBRIDGE_ROOT}/VERSION" HDRBRIDGE_VERSION LIMIT_COUNT 1)
set(PROJECT_VERSION "${HDRBRIDGE_VERSION}")
configure_file("${HDRBRIDGE_ROOT}/cmake/hdrbridge_version.h.in" "generated/hdrbridge_version.h" @ONLY)
find_package(libheif CONFIG REQUIRED)
find_package(libavif CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(PNG REQUIRED)
find_package(expat CONFIG REQUIRED)
find_package(JPEG REQUIRED)
find_package(TIFF REQUIRED)
find_package(LibLZMA REQUIRED)
find_package(lcms2 CONFIG REQUIRED)
find_package(JXR REQUIRED)
find_path(JXL_INCLUDE_DIR jxl/encode.h PATHS "${DEPS}/include" NO_DEFAULT_PATH REQUIRED)
find_library(JXL_LIBRARY NAMES jxl PATHS "${DEPS}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(JXL_CMS_LIBRARY NAMES jxl_cms PATHS "${DEPS}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(JXL_THREADS_LIBRARY NAMES jxl_threads PATHS "${DEPS}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(HIGHWAY_LIBRARY NAMES hwy PATHS "${DEPS}/lib" NO_DEFAULT_PATH REQUIRED)
foreach(BROTLI enc dec common)
  find_library(BROTLI_${BROTLI} NAMES brotli${BROTLI} PATHS "${DEPS}/lib" NO_DEFAULT_PATH REQUIRED)
endforeach()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(UHDR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(UHDR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(UHDR_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(UHDR_BUILD_FUZZERS OFF CACHE BOOL "" FORCE)
set(UHDR_BUILD_DEPS OFF CACHE BOOL "" FORCE)
set(UHDR_BUILD_JAVA OFF CACHE BOOL "" FORCE)
set(UHDR_ENABLE_HEIF OFF CACHE BOOL "" FORCE)
set(UHDR_ENABLE_GLES OFF CACHE BOOL "" FORCE)
set(UHDR_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(UHDR_ENABLE_INTRINSICS ON CACHE BOOL "" FORCE)
include("${HDRBRIDGE_ROOT}/cmake/PatchUltraHdr.cmake")
set(HDRBRIDGE_UHDR_ROOT "${HDRBRIDGE_ROOT}/.tools/libultrahdr")
if(EXISTS "${HDRBRIDGE_UHDR_ROOT}/CMakeLists.txt")
  hdrbridge_patch_ultrahdr("${HDRBRIDGE_UHDR_ROOT}")
  add_subdirectory("${HDRBRIDGE_UHDR_ROOT}" "libultrahdr" EXCLUDE_FROM_ALL)
else()
  include(FetchContent)
  FetchContent_Declare(libultrahdr GIT_REPOSITORY https://github.com/google/libultrahdr.git
    GIT_TAG b2aacb366e1542cfc29605cb0d8a0ebd06bb07f8
    PATCH_COMMAND "${CMAKE_COMMAND}" "-DHDRBRIDGE_UHDR_SOURCE=<SOURCE_DIR>" -P "${HDRBRIDGE_ROOT}/cmake/PatchUltraHdr.cmake")
  FetchContent_MakeAvailable(libultrahdr)
endif()
add_library(hdrbridge_jni SHARED bridge.cpp hdr_preview.cpp)
include("${HDRBRIDGE_ROOT}/cmake/HdrBridgeCoreSources.cmake")
hdrbridge_add_core_sources(hdrbridge_jni)
target_include_directories(hdrbridge_jni PRIVATE "${HDRBRIDGE_ROOT}/core/include"
  "${CMAKE_CURRENT_BINARY_DIR}/generated" "${JXL_INCLUDE_DIR}" ${JXR_INCLUDE_DIRS})
target_include_directories(hdrbridge_jni PRIVATE "${HDRBRIDGE_ROOT}/third_party/stb" "${HDRBRIDGE_ROOT}/third_party/nanosvg" "${HDRBRIDGE_UHDR_ROOT}/lib/include")
target_compile_definitions(hdrbridge_jni PRIVATE __ANSI__)
target_compile_options(hdrbridge_jni PRIVATE -O3 -fexceptions -frtti)
target_link_options(hdrbridge_jni PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--no-undefined")
target_link_libraries(hdrbridge_jni PRIVATE heif avif "${JXL_LIBRARY}" "${JXL_CMS_LIBRARY}"
  "${JXL_THREADS_LIBRARY}" "${HIGHWAY_LIBRARY}" "${BROTLI_enc}" "${BROTLI_dec}" "${BROTLI_common}"
  ${JXR_LIBRARIES} uhdr nlohmann_json::nlohmann_json PNG::PNG JPEG::JPEG TIFF::TIFF
  expat::expat LibLZMA::LibLZMA lcms2::lcms2 log android EGL GLESv3 dl)

option(HDRBRIDGE_ANDROID_CODEC_TESTS "Include device-only codec equivalence checks" OFF)
if(HDRBRIDGE_ANDROID_CODEC_TESTS)
  target_sources(hdrbridge_jni PRIVATE "${ANDROID_ROOT}/tests/codec-equivalence.cpp")
  target_include_directories(hdrbridge_jni PRIVATE "${HDRBRIDGE_UHDR_ROOT}/lib/include")
  target_compile_definitions(hdrbridge_jni PRIVATE UHDR_ENABLE_INTRINSICS)
endif()
